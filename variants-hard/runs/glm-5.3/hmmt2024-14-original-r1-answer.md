**Planteamiento:** Necesitamos contar caminos hamiltonianos en la retícula $3\times 24$ (72 puntos) que comienzan en $(2,1)$. Escribo un DFS con poda de conectividad:

- En cada paso, las celdas no visitadas deben formar un solo componente conexo y ese componente debe ser adyacente a la posición actual (si no, el camino no puede continuar).
- Poda adicional de paridad/gradu.

import sys
sys.setrecursionlimit(10000)

def count_paths(rows, cols, start):
    N = rows*cols
    FULL = (1 << N) - 1
    idx = lambda i,j: i*cols + j
    start_idx = idx(*start)
    # vecinos precomputados
    nbrs = [[] for _ in range(N)]
    for i in range(rows):
        for j in range(cols):
            for di,dj in ((1,0),(-1,0),(0,1),(0,-1)):
                ni,nj = i+di, j+dj
                if 0 <= ni < rows and 0 <= nj < cols:
                    nbrs[idx(i,j)].append(idx(ni,nj))

    from functools import lru_cache
    memo = {}
    def dfs(pos, mask):
        if mask == FULL:
            return 1
        key = (pos, mask)
        if key in memo:
            return memo[key]
        # marca la celda actual como visitada
        newmask = mask | (1 << pos)
        if newmask == FULL:
            return 1
        # poda: componentes conexos de no visitadas + adyacencia a pos
        unvis = FULL & ~newmask
        # BFS desde vecinos no visitados de pos
        seen = 0
        stack = [n for n in nbrs[pos] if not (newmask >> n) & 1]
        if not stack:
            return 0
        for s in stack: seen |= 1 << s
        while stack:
            c = stack.pop()
            for n in nbrs[c]:
                if not ((newmask >> n) & 1) and not ((seen >> n) & 1):
                    seen |= 1 << n
                    stack.append(n)
        if seen != unvis:
            memo[key] = 0
            return 0
        total = 0
        for n in nbrs[pos]:
            if not (newmask >> n) & 1:
                total += dfs(n, newmask)
        memo[key] = total
        return total

    return dfs(start_idx, 0)
